local function MakeDraggable(tb, ct)
    local d, di, ds, sp
    tb.InputBegan:Connect(function(inp)
        if inp.UserInputType == Enum.UserInputType.MouseButton1 or inp.UserInputType == Enum.UserInputType.Touch then
            d = true; ds = inp.Position; sp = ct.Position
            inp.Changed:Connect(function() if inp.UserInputState == Enum.UserInputState.End then d = false end end)
        end
    end)
    tb.InputChanged:Connect(function(inp)
        if inp.UserInputType == Enum.UserInputType.MouseMovement or inp.UserInputType == Enum.UserInputType.Touch then di = inp end
    end)
    UserInputService.InputChanged:Connect(function(inp)
        if inp == di and d then
            local dl = inp.Position - ds
            ct.Position = UDim2.new(sp.X.Scale, sp.X.Offset + dl.X, sp.Y.Scale, sp.Y.Offset + dl.Y)
        end
    end)
end

local function getMachineFromSeat(seat)
    if not seat then return nil end
    local cur = seat
    for _ = 1, 6 do
        if cur and cur.Parent and cur.Parent ~= workspace then
            cur = cur.Parent
            if cur:IsA("Model") then
                if cur:FindFirstChild("strengthGain") or cur:FindFirstChild("durabilityGain")
                   or cur:FindFirstChild("agilityGain") or cur:FindFirstChild("repTime")
                   or cur:FindFirstChild("interactSeat") then return cur end
            end
        end
    end
    return nil
end
local function findMachineBySeatPart(hum) if not hum then return nil end return hum.SeatPart and getMachineFromSeat(hum.SeatPart) or nil end
local function findMachineByWelded(char)
    if not char then return nil end
    for _, obj in ipairs(char:GetDescendants()) do
        if obj:IsA("Seat") or obj:IsA("VehicleSeat") then local m = getMachineFromSeat(obj) if m then return m end end
    end
    return nil
end
local function findMachineByProximity(hrp, radius)
    if not hrp then return nil end
    radius = radius or 25
    local best, bestDist = nil, radius
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("Model") then
            local hasGain = obj:FindFirstChild("strengthGain") or obj:FindFirstChild("durabilityGain") or obj:FindFirstChild("repTime")
            if hasGain then
                local seat = obj:FindFirstChildWhichIsA("Seat", true) or obj:FindFirstChildWhichIsA("VehicleSeat", true)
                local checkPos = seat and seat.Position or (obj.PrimaryPart and obj.PrimaryPart.Position)
                if checkPos then
                    local d = (checkPos - hrp.Position).Magnitude
                    if d < bestDist then bestDist = d; best = obj end
                end
            end
        end
    end
    return best
end
local function findMachine()
    local char = Player.Character
    if not char then return nil end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end
    local m = findMachineBySeatPart(hum) if m then return m, "SeatPart" end
    m = findMachineByWelded(char) if m then return m, "Welded" end
    if hum and hum.Sit then m = findMachineByProximity(hrp, 15) if m then return m, "Proximity" end end
    return nil
end

local AutoFast = {
    active = false, mode = "none", target = nil, detectMethod = "",
    machine = nil, tool = nil, machineRemotes = {}, toolRemotes = {},
    cps = 0, totalCalls = 0, tickRate = 0.0001,

local function cacheMachineRemotes(m)
    if not m or AutoFast.machine == m then return end
    AutoFast.machine = m
    AutoFast.machineRemotes = {}
    for _, obj in ipairs(m:GetDescendants()) do
        if obj:IsA("RemoteEvent") or obj:IsA("RemoteFunction") then table.insert(AutoFast.machineRemotes, obj) end
    end
end
local function cacheToolRemotes(t)
    if not t or AutoFast.tool == t then return end
    AutoFast.tool = t
    AutoFast.toolRemotes = {}
    for _, obj in ipairs(t:GetDescendants()) do
        if obj:IsA("RemoteEvent") or obj:IsA("RemoteFunction") then table.insert(AutoFast.toolRemotes, obj) end
    end
end
local function fireMachine(m, hum)
    if not m then return end
    pcall(function() hum.Sit = false end)
    pcall(function() hum.Sit = true end)
    local rep = m:FindFirstChild("repTime")
    if rep then pcall(function() rep.Value = 0 end) end
    for _, nm in ipairs({"strengthGain","durabilityGain","agilityGain"}) do
        local v = m:FindFirstChild(nm)
        if v then pcall(function() v.Value = 999999999 end) end
    end
    for i = 1, #AutoFast.machineRemotes do
        local r = AutoFast.machineRemotes[i]
        pcall(function()
            if r:IsA("RemoteEvent") then r:FireServer() r:FireServer(0) r:FireServer(1) r:FireServer(true) r:FireServer(hum)
            else r:InvokeServer() end
        end)
    end
    for _, obj in ipairs(m:GetDescendants()) do
        if obj:IsA("ProximityPrompt") then
            obj.HoldDuration = 0
            obj.MaxActivationDistance = 999
            if type(fireproximityprompt) == "function" then pcall(fireproximityprompt, obj, 0) end
        end
    end
    AutoFast.cps = AutoFast.cps + 1
    AutoFast.totalCalls = AutoFast.totalCalls + 1
end
local function fireTool(t)
    if not t then return end
    for i = 1, #AutoFast.toolRemotes do
        local r = AutoFast.toolRemotes[i]
        pcall(function()
            if r:IsA("RemoteEvent") then r:FireServer() r:FireServer(0) r:FireServer(t)
            else r:InvokeServer() end
        end)
    end
    pcall(function() t.Enabled = true end)
    pcall(function() t:Activate() end)
    virtualClick()
    AutoFast.cps = AutoFast.cps + 1
    AutoFast.totalCalls = AutoFast.totalCalls + 1
end
local function autoFastTick()
    if not AutoFast.active then return end
    local char = Player.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then return end
    local m, method = findMachine()
    if m then
        AutoFast.mode = "seat"; AutoFast.target = m.Name; AutoFast.detectMethod = method or "?"
        cacheMachineRemotes(m); killTrainingAnims(hum); fireMachine(m, hum)
        return
    end
    local tool = char:FindFirstChildOfClass("Tool")
    if tool and not isPunchTool(tool) then
        AutoFast.mode = "tool"; AutoFast.target = tool.Name; AutoFast.detectMethod = "Tool"
        cacheToolRemotes(tool); killTrainingAnims(hum); fireTool(tool)
        return
    end
    AutoFast.mode = "none"; AutoFast.target = nil; AutoFast.detectMethod = ""
    AutoFast.machine = nil; AutoFast.tool = nil
end

    end
end)

        return false
    end
    for _, obj in ipairs(workspace:GetDescendants()) do

    local arena = ev:FindFirstChild("BossArena")
    if not arena then return nil, nil, nil end
    for i = 1, 6 do
        local slot = arena:FindFirstChild("Boss" .. i)
        if slot and isBossAliveInSlot(slot) then
            local bp = getBossBodyPart(slot)
            if bp then return bp, makeFakeHumanoid(slot), slot.Name end
        end
    end
    return nil, nil, nil
end
local function findChest()
    local arenaPos = bossFarm["AutoBoss"].arenaPos or ARENA_FALLBACK
    local best, bestDist = nil, 300
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("BasePart") or obj:IsA("Model") then
            local n = string.lower(obj.Name)
            local isChest = false
            for _, kw in ipairs(CHEST_KEYWORDS) do if n:find(kw, 1, true) then isChest = true break end end
            if isChest then
                local pos
                if obj:IsA("BasePart") then pos = obj.Position
                else for _, d in ipairs(obj:GetDescendants()) do if d:IsA("BasePart") then pos = d.Position break end end end
                if pos then
                    local d = (pos - arenaPos).Magnitude
                    if d < bestDist then bestDist = d; best = obj end
                end
            end
        end
    end
    return best
end
local function getPunchTool()
    local char = Player.Character
    if not char then return nil end
    local bp = Player:FindFirstChild("Backpack")
    for _, t in ipairs(char:GetChildren()) do if t:IsA("Tool") and isPunchTool(t) then return t end end
    if bp then for _, t in ipairs(bp:GetChildren()) do if t:IsA("Tool") and isPunchTool(t) then return t end end end
    return nil
end
local function createHitbox()
    local data = bossFarm["AutoBoss"]
    if data.hitboxPart and data.hitboxPart.Parent then return data.hitboxPart end
    local char = Player.Character
    if not char then return nil end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end
    local hitbox = Instance.new("Part")
    hitbox.Name = "PS_BossHitbox"
    hitbox.Size = Vector3.new(30, 30, 30)
    hitbox.Transparency = 1
    hitbox.CanCollide = false
    hitbox.Massless = true
    hitbox.Anchored = false
    hitbox.CFrame = hrp.CFrame
    hitbox.Parent = char
    local weld = Instance.new("WeldConstraint")
    weld.Part0 = hrp
    weld.Part1 = hitbox
    weld.Parent = hitbox
    data.hitboxPart = hitbox
    return hitbox
end
local function destroyHitbox()
    local data = bossFarm["AutoBoss"]
    if data.hitboxPart then
        pcall(function() data.hitboxPart:Destroy() end)
        data.hitboxPart = nil
    end
end

task.spawn(function()
    while true do
        local data = bossFarm["AutoBoss"]
        if data and data.active then
            if not (data.hitboxPart and data.hitboxPart.Parent) then createHitbox() end
            local char = Player.Character
            local myHrp = char and char:FindFirstChild("HumanoidRootPart")
            data.arenaPos = getArenaPosition()
            local needsTP = false
            if myHrp then
                local distToArena = (myHrp.Position - data.arenaPos).Magnitude
                if distToArena > 40 then needsTP = true end
            end
            if data.mode == "teleporting" then
                if myHrp then
                    pcall(function()
                        myHrp.CFrame = CFrame.new(data.arenaPos)
                        myHrp.Velocity = Vector3.new(0, 0, 0)
                        myHrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                    end)
                end
                data.mode = "waiting_spawn"
            end
            if data.mode == "waiting_spawn" then
                if needsTP and myHrp then
                    pcall(function()
                        myHrp.CFrame = CFrame.new(data.arenaPos)
                        myHrp.Velocity = Vector3.new(0, 0, 0)
                        myHrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                    end)
                end
                local bHrp, bHum, slot = findAnyBossAlive()
                if bHrp and bHum then
                    data.target = bHrp; data.targetHum = bHum; data.slotName = slot
                    data.mode = "fighting"
                end
            end
            if data.mode == "fighting" then
                local hrp = data.target
                local alive = false
                if hrp and hrp.Parent then
                    local hp = getBossHealthAttr(hrp.Parent)
                    if hp ~= nil then alive = hp > 0 end
                end
                if not alive then
                    data.mode = "waiting_chest"
                    data.chestWaitStart = tick()
                    data.target = nil
                    data.targetHum = nil
                end
            end
            if data.mode == "waiting_chest" then
                local chest = findChest()
                if chest then
                    data.target = chest
                    data.mode = "opening_chest"
                else
                    if tick() - (data.chestWaitStart or 0) > 15 then
                        data.mode = "teleporting"
                        data.target = nil
                        data.targetHum = nil
                    end
                end
            end
            if data.mode == "opening_chest" then
                local chest = data.target
                if not chest or not chest.Parent then
                    data.mode = "teleporting"
                    data.target = nil
                    data.targetHum = nil
                end
            end
        else
            if bossFarm["AutoBoss"].hitboxPart then destroyHitbox() end
        end
        task.wait(SEARCH_RATE)
    end
end)

RunService.RenderStepped:Connect(function()
    local char = Player.Character
    if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    local data = bossFarm["AutoBoss"]
    if not data or not data.active or not data.target then return end
    if lockPosState.active then return end
    if data.mode == "fighting" then
        local body = data.target
        if body and body.Parent then
            local distance = 15
            local yOffset = (body.Size.Y / 2) + HEAD_OFFSET
            local myPos = hrp.Position
            local bossPos = body.Position
            local dir = (myPos - bossPos)
            if dir.Magnitude < 0.1 then dir = Vector3.new(0, 0, 1) end
            dir = dir.Unit
            local targetPos = bossPos + dir * distance + Vector3.new(0, yOffset, 0)
            pcall(function()
                hrp.CFrame = CFrame.lookAt(targetPos, bossPos)
                hrp.Velocity = Vector3.new(0, 0, 0)
                hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
            end)
        end
    elseif data.mode == "opening_chest" then
        local chest = data.target
        if chest and chest.Parent then
            local cpos
            if chest:IsA("BasePart") then cpos = chest.Position
            else for _, d in ipairs(chest:GetDescendants()) do if d:IsA("BasePart") then cpos = d.Position break end end end
            if cpos then
                pcall(function()
                    hrp.CFrame = CFrame.new(cpos + Vector3.new(0, 4, 0))
                    hrp.Velocity = Vector3.new(0, 0, 0)
                    hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                end)
            end
        end
    end
end)

    local function gSV(tp, kws)
        if not tp or not tp.Parent then return "N/A" end
        local function sc(c)
            if not c then return nil end
            for _, it in ipairs(c:GetChildren()) do
                local n = string.lower(it.Name)
                for _, kw in ipairs(kws) do if n:find(string.lower(kw)) and it:IsA("ValueBase") then return tostring(it.Value) end end
            end
            return nil
        end
        local ls = tp:FindFirstChild("leaderstats")

    local function findGlobal(kw)
        local myC = Player.Character
        local lkw = string.lower(kw)
        local best, bs = nil, 0
        local BAD = {"tobeach","tospawn","press","machine","frame","cylinder","npc","lefthand","righthand","leftarm","rightarm","head","mainglass","maingem","crystal","orb","chest","trigger","hitbox","zone"}
        local function hasBad(s)
            if not s then return false end
            local l = string.lower(s)
            for _, w in ipairs(BAD) do if l:find(w, 1, true) then return true end end
            return false
        end
        for _, obj in ipairs(workspace:GetDescendants()) do
            if obj:IsA("BasePart") and not (myC and obj:IsDescendantOf(myC)) then
                local fp = obj:GetFullName()
                local ln = string.lower(obj.Name)
                local lp = string.lower(fp)
                if not hasBad(ln) and not hasBad(lp) then
                    if ln:find(lkw, 1, true) or lp:find(lkw, 1, true) then
                        local sc = #lkw
                        if ln:find("islandmodel", 1, true) then sc = sc + 200 end
                        if ln:find("floor", 1, true) then sc = sc + 80 end
                        if lp:find("island", 1, true) then sc = sc + 50 end
                        if sc > bs then bs = sc best = obj end
                    end
                end
            end
        end
        return best
    end
